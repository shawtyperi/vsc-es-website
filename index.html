// [ฟังก์ชันสําคัญ] สำหรับกดเรียกใช้ (Run) ครั้งแรก เพื่อให้ระบบขออนุมัติสิทธิ์เข้าใช้งาน Google Drive & ส่งเมล Gmail
function runFirstTimeAuthorization() {
  var folderName = "VSC Empeo Screenshots";
  var folders = DriveApp.getFoldersByName(folderName);
  var folder;
  if (folders.hasNext()) {
    folder = folders.next();
  } else {
    folder = DriveApp.createFolder(folderName);
  }
  
  // ตรวจสอบและสร้างหน้า "Email-Active" ไว้รองรับผู้สมัคร
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Email-Active");
  if (!sheet) {
    sheet = ss.insertSheet("Email-Active");
    sheet.appendRow(["อีเมล", "รหัสผ่าน", "ชื่อ-นามสกุล", "ชื่อเล่น", "ตำแหน่ง", "วันที่สมัคร"]);
  }
  Logger.log("อนุมัติสิทธิ์สำเร็จแล้ว! โฟลเดอร์ปลายทางใน Drive คือ: " + folder.getName());
}

// ฟังก์ชันสำหรับดึงสถานะการแก้ไขกลับไปแสดงที่พนักงานแบบอัตโนมัติ (doGet)
function doGet(e) {
  if (e.parameter.action === "checkStatus" && e.parameter.ids) {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheets()[0];
      var data = sheet.getDataRange().getValues();
      
      var requestedIds = e.parameter.ids.split(",");
      var results = {};
      
      // ค้นหาเรื่องแจ้งที่ระบุ ID
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var rowId = row[10] ? row[10].toString().trim() : ""; // คอลัมน์ K (รหัสอ้างอิง)
        
        // ค้นหาเผื่อแอดมินสลับคอลัมน์
        if (!rowId) {
          for (var col = 0; col < row.length; col++) {
            if (row[col] && row[col].toString().indexOf("REP-") === 0) {
              rowId = row[col].toString().trim();
              break;
            }
          }
        }
        
        if (rowId && requestedIds.indexOf(rowId) !== -1) {
          var columnJValue = row[9] ? row[9].toString().trim() : ""; // คอลัมน์ J (ผลการแก้ไข)
          var columnHValue = row[7] ? row[7].toString().trim() : ""; // คอลัมน์ H (สถานะ)
          
          var statusText = "รอดำเนินการ";
          // ถ้ามีคำว่า "แก้ไขแล้ว" ในคอลัมน์ J หรือคอลัมน์ H ให้เป็น ดำเนินการเรียบร้อย
          if (columnJValue.indexOf("แก้ไขแล้ว") !== -1 || columnHValue.indexOf("แก้ไขแล้ว") !== -1 || columnJValue.indexOf("ดำเนินการเรียบร้อย") !== -1) {
            statusText = "ดำเนินการเรียบร้อย";
          } else if (columnJValue) {
            statusText = columnJValue; // โชว์ข้อความที่แอดมินพิมพ์ใน J
          } else if (columnHValue) {
            statusText = columnHValue;
          }
          
          results[rowId] = statusText;
        }
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "success",
        "results": results
      }))
      .setMimeType(ContentService.MimeType.JSON);
      
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({
        "status": "error",
        "message": err.toString()
      }))
      .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // รองรับการ Login และดึงประวัติของพนักงานเฉพาะคนผ่าน doGet
  if (e.parameter.action === "getHistory" && e.parameter.email) {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheets()[0];
      var data = sheet.getDataRange().getValues();
      
      var targetEmail = e.parameter.email.toString().trim().toLowerCase();
      var userHistory = [];
      
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var rowEmail = row[11] ? row[11].toString().trim().toLowerCase() : "";
        
        if (rowEmail === targetEmail) {
          var columnJValue = row[9] ? row[9].toString().trim() : "";
          var columnHValue = row[7] ? row[7].toString().trim() : "";
          
          var statusText = "รอดำเนินการ";
          if (columnJValue.indexOf("แก้ไขแล้ว") !== -1 || columnHValue.indexOf("แก้ไขแล้ว") !== -1 || columnJValue.indexOf("ดำเนินการเรียบร้อย") !== -1) {
            statusText = "ดำเนินการเรียบร้อย";
          } else if (columnJValue) {
            statusText = columnJValue;
          } else if (columnHValue) {
            statusText = columnHValue;
          }

          userHistory.push({
            id: row[10] ? row[10].toString().trim() : "REP-" + i,
            issueType: row[4] ? row[4].toString() : "",
            issueDate: row[5] ? row[5].toString() : "",
            createdTime: row[0] ? row[0].toString() : "",
            status: statusText,
            syncStatus: "Synced"
          });
        }
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "success",
        "history": userHistory
      }))
      .setMimeType(ContentService.MimeType.JSON);
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({
        "status": "error",
        "message": err.toString()
      }))
      .setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    "status": "success",
    "message": "ระบบ VSC - ES (Empeo Support) API พร้อมทำงานบนตารางงานปัจจุบันแล้ว!"
  }))
  .setMimeType(ContentService.MimeType.JSON);
}

// ค้นหาแถวของอีเมลในชีท Email-Active (ส่งกลับดัชนีแถวแบบ 1-Based)
function findEmailRow(email, ss) {
  var sheet = ss.getSheetByName("Email-Active");
  if (!sheet) {
    return -1;
  }
  var data = sheet.getDataRange().getValues();
  email = email.trim().toLowerCase();
  for (var i = 0; i < data.length; i++) {
    if (data[i][0] && data[i][0].toString().trim().toLowerCase() === email) {
      return i + 1; // 1-based row index
    }
  }
  return -1;
}

// ฟังก์ชันส่งเมลแบบป้องกันล้มเหลว (Safe Mail Sender)
function sendEmailWithFallback(email, subject, bodyText) {
  try {
    MailApp.sendEmail({
      to: email,
      subject: subject,
      body: bodyText
    });
    return true;
  } catch (err1) {
    try {
      GmailApp.sendEmail(email, subject, bodyText);
      return true;
    } catch (err2) {
      Logger.log("ไม่สามารถส่งอีเมลได้: " + err2.toString());
      throw new Error("ระบบบล็อกการส่งเมลของ Google ชั่วคราว กรุณาติดต่อ HR/IT: " + err2.toString());
    }
  }
}

// ฟังก์ชันหลักในการรับข้อมูลปัญหาและรูปภาพหน้าจอ
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // A. ส่ง OTP สำหรับการลงทะเบียน (Register OTP)
    if (data.action === "sendRegisterOTP") {
      var email = data.email.trim().toLowerCase();
      
      var sheet = ss.getSheetByName("Email-Active");
      if (!sheet) {
        sheet = ss.insertSheet("Email-Active");
        sheet.appendRow(["อีเมล", "รหัสผ่าน", "ชื่อ-นามสกุล", "ชื่อเล่น", "ตำแหน่ง", "วันที่สมัคร"]);
      }
      
      var emailRow = findEmailRow(email, ss);
      if (emailRow !== -1) {
        var existingPassword = sheet.getRange(emailRow, 2).getValue().toString().trim();
        if (existingPassword !== "") {
          return ContentService.createTextOutput(JSON.stringify({
            "status": "error",
            "message": "อีเมลนี้ผ่านการลงทะเบียนเข้าใช้เรียบร้อยแล้ว ท่านสามารถล็อกอินได้ทันที"
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      
      // สร้าง OTP 6 หลัก และเก็บใน cache (มีอายุ 10 นาที)
      var otp = Math.floor(100000 + Math.random() * 900000).toString();
      var cache = CacheService.getScriptCache();
      cache.put("OTP_REG_" + email, otp, 600); // 10 นาที
      
      // ส่งอีเมลหาพนักงานโดยตรง
      var subject = "รหัส OTP สำหรับยืนยันตัวตนระบบ VSC - ES (Empeo Support)";
      var bodyText = "สวัสดีครับ,\n\nรหัสยืนยันตัวตน (OTP) สำหรับเปิดใช้งานบัญชี VSC - ES (Empeo Support) ของคุณคือ: " + otp + "\n\nรหัสนี้มีอายุใช้งาน 10 นาที หากคุณไม่ได้ส่งคำขอนี้ กรุณาติดต่อแอดมิน\n\nด้วยความเคารพ,\nฝ่ายทรัพยากรบุคคล (HR)";
      
      sendEmailWithFallback(email, subject, bodyText);
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "success",
        "message": "ส่งรหัส OTP ไปที่อีเมลของท่านเรียบร้อยแล้ว กรุณาเปิดกล่องข้อความตรวจสอบ"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // B. ฟังก์ชันการลงทะเบียนพนักงานใหม่ (Register) -> สมัครได้ทุกคนแล้วเอาข้อมูลบันทึกใน Email-Active
    if (data.action === "register") {
      var email = data.email.trim().toLowerCase();
      var otpSubmitted = data.otp.toString().trim();
      
      // ตรวจสอบ OTP ใน Cache
      var cache = CacheService.getScriptCache();
      var cachedOtp = cache.get("OTP_REG_" + email);
      if (!cachedOtp || cachedOtp !== otpSubmitted) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "รหัส OTP ไม่ถูกต้อง หรือหมดอายุแล้ว กรุณากดขอ OTP ใหม่อีกครั้ง"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var sheet = ss.getSheetByName("Email-Active");
      if (!sheet) {
        sheet = ss.insertSheet("Email-Active");
        sheet.appendRow(["อีเมล", "รหัสผ่าน", "ชื่อ-นามสกุล", "ชื่อเล่น", "ตำแหน่ง", "วันที่สมัคร"]);
      }
      
      var emailRow = findEmailRow(email, ss);
      
      if (emailRow !== -1) {
        // มีแถวอีเมลนี้อยู่แล้ว (เช่น แอดมินสร้างไว้แต่แรก) ให้เขียนรหัสผ่านและข้อมูลทับ
        sheet.getRange(emailRow, 2).setValue(data.password.toString().trim());
        sheet.getRange(emailRow, 3).setValue(data.name.toString().trim());
        sheet.getRange(emailRow, 4).setValue(data.nickname.toString().trim());
        sheet.getRange(emailRow, 5).setValue(data.position.toString().trim());
        sheet.getRange(emailRow, 6).setValue(new Date());
      } else {
        // ยังไม่มีข้อมูลอีเมลนี้เลย ให้ Append แถวใหม่ทันที
        sheet.appendRow([
          email,
          data.password.toString().trim(),
          data.name.toString().trim(),
          data.nickname.toString().trim(),
          data.position.toString().trim(),
          new Date()
        ]);
      }
      
      // ล้าง OTP ออกหลังจากใช้งานแล้ว
      cache.remove("OTP_REG_" + email);
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "success",
        "message": "ลงทะเบียนพนักงานสำเร็จแล้ว!"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // C. ส่ง OTP สำหรับเปลี่ยนรหัสผ่าน (Reset Password OTP)
    if (data.action === "sendResetOTP") {
      var email = data.email.trim().toLowerCase();
      
      var sheet = ss.getSheetByName("Email-Active");
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "ระบบเก็บฐานข้อมูลยังไม่ได้เริ่มต้นสมัครใช้งาน (ไม่พบชีท Email-Active)"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var emailRow = findEmailRow(email, ss);
      if (emailRow === -1) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "ไม่พบอีเมลนี้ในระบบสมัครใช้งาน กรุณาลงทะเบียนสมาชิกใหม่ก่อนครับ"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var existingPassword = sheet.getRange(emailRow, 2).getValue().toString().trim();
      if (existingPassword === "") {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "อีเมลนี้ยังไม่ได้ลงทะเบียนตั้งรหัสผ่าน กรุณาทำรายการสมัครสมาชิกใหม่ก่อนครับ"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var otp = Math.floor(100000 + Math.random() * 900000).toString();
      var cache = CacheService.getScriptCache();
      cache.put("OTP_RESET_" + email, otp, 600);
      
      var subject = "รหัส OTP สำหรับเปลี่ยนรหัสผ่าน VSC - ES (Empeo Support)";
      var bodyText = "สวัสดีครับ,\n\nรหัสยืนยันตัวตน (OTP) สำหรับเปลี่ยนรหัสผ่านบัญชี VSC - ES (Empeo Support) ของคุณคือ: " + otp + "\n\nรหัสนี้มีอายุใช้งาน 10 นาที\n\nด้วยความเคารพ,\nฝ่ายทรัพยากรบุคคล (HR)";
      
      sendEmailWithFallback(email, subject, bodyText);
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "success",
        "message": "ส่งรหัส OTP สำหรับเปลี่ยนรหัสผ่านสำเร็จแล้ว!"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // D. ยืนยันรหัสผ่านใหม่ (Reset Password Confirm) -> อัปเดตใน Email-Active
    if (data.action === "resetPassword") {
      var email = data.email.trim().toLowerCase();
      var otpSubmitted = data.otp.toString().trim();
      var newPassword = data.newPassword.toString().trim();
      
      var cache = CacheService.getScriptCache();
      var cachedOtp = cache.get("OTP_RESET_" + email);
      if (!cachedOtp || cachedOtp !== otpSubmitted) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "รหัส OTP ไม่ถูกต้อง หรือหมดอายุแล้ว"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var emailRow = findEmailRow(email, ss);
      if (emailRow !== -1) {
        var sheet = ss.getSheetByName("Email-Active");
        sheet.getRange(emailRow, 2).setValue(newPassword);
        cache.remove("OTP_RESET_" + email);
        return ContentService.createTextOutput(JSON.stringify({
          "status": "success",
          "message": "เปลี่ยนรหัสผ่านใหม่สำเร็จแล้ว!"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "error",
        "message": "ไม่พบพนักงานคนดังกล่าวในระบบจัดเก็บฐานข้อมูล"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // E. ฟังก์ชันการเข้าสู่ระบบพนักงาน (Login)
    if (data.action === "login") {
      var email = data.email.trim().toLowerCase();
      var password = data.password.toString().trim();
      
      var emailRow = findEmailRow(email, ss);
      if (emailRow === -1) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "ไม่พบอีเมลนี้ในระบบฐานข้อมูล กรุณาคลิกสมัครสมาชิกใหม่ก่อนเข้าใช้งานครับ"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var sheet = ss.getSheetByName("Email-Active");
      var savedPassword = sheet.getRange(emailRow, 2).getValue().toString().trim();
      var name = sheet.getRange(emailRow, 3).getValue().toString().trim();
      var nickname = sheet.getRange(emailRow, 4).getValue().toString().trim();
      var position = sheet.getRange(emailRow, 5).getValue().toString().trim();
      
      if (savedPassword === "") {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error",
          "message": "อีเมลนี้ยังไม่ได้ลงทะเบียนเข้าใช้งาน กรุณากดลงทะเบียนสมาชิกใหม่ก่อนเข้าใช้งานครั้งแรกครับ"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      if (savedPassword === password) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "success",
          "user": {
            "email": email,
            "name": name,
            "nickname": nickname,
            "position": position
          }
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        "status": "error",
        "message": "อีเมลหรือรหัสผ่านไม่ถูกต้อง กรุณาตรวจสอบใหม่อีกครั้ง"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // F. บันทึกรายงานปัญหาของพนักงาน (Submit Report)
    var fileUrlsList = [];
    
    // จัดการแยกและแปลงรูปภาพ Base64
    if (data.screenshots && data.screenshots.length > 0) {
      for (var i = 0; i < data.screenshots.length; i++) {
        var base64Img = data.screenshots[i];
        if (base64Img && base64Img.indexOf("data:image") === 0) {
          try {
            var splitData = base64Img.split(",");
            var contentType = splitData[0].split(":")[1].split(";")[0];
            var base64Data = splitData[1];
            
            var decoded = Utilities.base64Decode(base64Data);
            var fileExtension = contentType.split("/")[1] || "png";
            
            var fileName = "screenshot_" + data.nickname + "_" + i + "_" + new Date().getTime() + "." + fileExtension;
            var blob = Utilities.newBlob(decoded, contentType, fileName);
            
            var folderName = "VSC Empeo Screenshots";
            var folders = DriveApp.getFoldersByName(folderName);
            var folder;
            if (folders.hasNext()) {
              folder = folders.next();
            } else {
              folder = DriveApp.createFolder(folderName);
            }
            
            var createdFile = folder.createFile(blob);
            createdFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            fileUrlsList.push(createdFile.getUrl());
            
          } catch (fileError) {
            fileUrlsList.push("เกิดปัญหาการบันทึกภาพรูปที่ " + (i+1) + ": " + fileError.toString());
          }
        }
      }
    }
    
    var finalFileUrls = fileUrlsList.length > 0 ? fileUrlsList.join("\n") : "ไม่มีรูปภาพแนบ";
    
    var sheet = ss.getSheets()[0]; 
    var timestamp = new Date();
    var formattedDate = Utilities.formatDate(timestamp, "GMT+7", "yyyy-MM-dd HH:mm:ss");
    
    sheet.appendRow([
      formattedDate,
      data.employeeName,
      data.nickname,
      data.position,
      data.issueType,
      data.issueDate,
      data.description,
      data.status || "รอดำเนินการ",
      finalFileUrls,
      "",
      data.id,
      data.email
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      "status": "success",
      "message": "บันทึกข้อมูลและอัปโหลดรูปหลักฐานลง Google Drive ของคุณเรียบร้อยแล้ว!"
    }))
    .setMimeType(ContentService.MimeType.JSON);
    
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error",
      "message": error.toString()
    }))
    .setMimeType(ContentService.MimeType.JSON);
  }
}
